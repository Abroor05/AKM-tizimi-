🔐 Auth Middleware (auth.js)

JWT token generatsiya, verify, role-based access control

// JWT konfiguratsiya
const JWT_SECRET = 'kbt-secret-key-2024';
const JWT_EXPIRES_IN = '24h';

// Token generatsiya
export function generateToken(user) {
  return jwt.sign(
    { id: user.id, username: user.username, role: user.role },
    JWT_SECRET,
    { expiresIn: JWT_EXPIRES_IN }
  );
}

// Middleware — har bir so'rovda token tekshiradi
export function authMiddleware(req, res, next) {
  const header = req.headers.authorization;
  if (!header || !header.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Token talab qilinadi' });
  }
  const token = header.slice(7);
  const decoded = jwt.verify(token, JWT_SECRET);
  const user = getById('users', decoded.id);
  req.user = user;  // foydalanuvchi so'rovga biriktiriladi
  next();
}

// Rol asosida himoya
export function requireRole(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Ruxsat berilmagan' });
    }
    next();
  };
}

🔄 Generic CRUD Factory (crud.js)

9 ta entity uchun bir xil CRUD route'lari avtomatik yaratiladi

// Entity konfiguratsiya — API path -> DB jadval
const ENTITY_CONFIG = {
  'libraries': 'libraries',
  'books': 'books',
  'readers': 'readers',
  'activities': 'activities',
  'events': 'events',
  'inventory': 'inventory',
  'documents': 'documents',
  'appeals': 'appeals',
  'kpi-data': 'kpi_data',
};

// Har bir entity uchun avtomatik route yaratish
export function createCrudRouter(apiPath, tableName) {
  const router = Router();

  router.get('/', authMiddleware, (req, res) => {        // GET — ro'yxat
    res.json(getAll(tableName));
  });

  router.get('/:id', authMiddleware, (req, res) => {     // GET — bittasi
    res.json(getById(tableName, req.params.id));
  });

  router.post('/', authMiddleware, (req, res) => {        // POST — yaratish
    const id = generateId(prefix);
    insertRow(tableName, { ...req.body, id });
    res.status(201).json(getById(tableName, id));
  });

  router.put('/:id', authMiddleware, (req, res) => {      // PUT — yangilash
    res.json(updateRow(tableName, req.params.id, req.body));
  });

  router.delete('/:id', authMiddleware, (req, res) => {   // DELETE — o'chirish
    deleteRow(tableName, req.params.id);
    res.json({ success: true });
  });

  return router;
}

// 9 ta entity uchun route'larni app'ga biriktirish
export function mountCrudRoutes(app) {
  for (const [apiPath, tableName] of Object.entries(ENTITY_CONFIG)) {
    app.use(`/api/${apiPath}`, createCrudRouter(apiPath, tableName));
  }
}
